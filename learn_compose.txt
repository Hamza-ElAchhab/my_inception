

Command	الوظيفة
docker compose build	بناء جميع Images
docker compose build nginx or more	بناء Image ديال service محدد



Command	الوظيفة
docker compose up	بناء (إذا لزم) وتشغيل Services
docker compose up -d	التشغيل في Background
docker compose up --build	إعادة البناء ثم التشغيل
docker compose up -d --build	البناء والتشغيل في Background
docker compose up nginx	تشغيل Service محدد


Command	الوظيفة
docker compose down	إيقاف وحذف Containers وNetworks ديال Compose
docker compose down --volumes	نفس الشيء مع حذف Volumes
docker compose down --rmi all	حذف Images المستعملة


Command	الوظيفة
docker compose start	تشغيل Containers متوقفة
docker compose stop	إيقاف Containers بدون حذفها
docker compose restart	إعادة تشغيل جميع Services
docker compose restart mariadb	إعادة تشغيل MariaDB
